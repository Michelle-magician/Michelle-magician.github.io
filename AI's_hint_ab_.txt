// 1. カメラのステート（中心座標とズーム倍率）
const camera = {
    x: 400,   // カメラが映したい中心のX座標（例：プレイヤーのX）
    y: 300,   // カメラが映したい中心のY座標（例：プレイヤーのY）
    zoom: 1.0 // ズーム倍率（1.0が等倍）
};

// 2. 描画するオブジェクトの配列（ご提示の形式）
// [type, x, y, w, h, angle(度), resource]
const renderQueue = [
    ['player', 400, 300, 50, 50, 0, imgPlayer],
    ['enemy',  550, 200, 40, 40, 45, imgEnemy],
    ['item',   300, 400, 20, 20, -10, imgItem]
];

function drawScene(ctx, canvas, camera, objects) {
    // 画面を一度クリア
    ctx.clearRect(0, 0, canvas.width, canvas.height);

    // --- 【重要】カメラのトランスフォーム適用 ---
    ctx.save();

    // ステップA: カメラの中心が、画面（キャンバス）のちょうど真ん中に来るように調整
    ctx.translate(canvas.width / 2, canvas.height / 2);
    
    // ステップB: ズームを適用
    ctx.scale(camera.zoom, camera.zoom);
    
    // ステップC: カメラの座標分、世界を「逆方向」に引っ張る
    ctx.translate(-camera.x, -camera.y);
    // -------------------------------------------

    // 3. 配列をループしてすべてのオブジェクトを描画
    for (const obj of objects) {
        const [type, x, y, w, h, angle, resource] = obj;

        // 画像がまだ読み込まれていない場合はスキップ
        if (!resource || !resource.complete) continue;

        ctx.save();
        
        // オブジェクトの「中心」を原点にする
        // （すでにカメラ分は上で引かれているので、ここでは純粋なワールド座標を渡すだけ！）
        ctx.translate(x, y);
        
        // 回転（度をラジアンに変換）
        if (angle !== 0) {
            ctx.rotate(angle * Math.PI / 180);
        }

        // 中心からずらして描画（2Pカラーなどの特殊処理もここに挟めます）
        ctx.drawImage(resource, -w / 2, -h / 2, w, h);

        ctx.restore();
    }

    // カメラのトランスフォームを解除（UIやパラメーターなどを等倍・固定位置で描くため）
    ctx.restore();
    
    // ※もしカメラに追従しない「固定のUI（HPゲージやボタン）」を描くなら、ここに書きます
    // ctx.fillText("Score: 100", 20, 20); 
}
